```mermaid
    }

    }
    
    %% Табличные части

    }
    
    РеализацияТоваровУслуг_Товары {

        number Цена
        number Сумма
    }
    
    %% Регистры
    РегистрСведений_ОстаткиТоваров {

    Организации ||--o{ ПоступлениеНаРасчетныйСчет : "получает деньги"
    Организации ||--o{ СписаниеСРасчетногоСчета : "платит деньги"
    
    Контрагенты ||--o{ ПоступлениеТоваровУслуг : "поставляет товары"
    Контрагенты ||--o{ РеализацияТоваровУслуг : "покупает товары"
    Контрагенты ||--o{ ПоступлениеНаРасчетныйСчет : "переводит деньги"
    Контрагенты ||--o{ СписаниеСРасчетногоСчета : "получает платежи"
    
    Номенклатура ||--o{ ПоступлениеТоваровУслуг_Товары : "поступает"
    Номенклатура ||--o{ РеализацияТоваровУслуг_Товары : "продается"
    Номенклатура ||--o{ РегистрСведений_ОстаткиТоваров : "хранится"
    
    ПоступлениеТоваровУслуг ||--|{ ПоступлениеТоваровУслуг_Товары : "содержит товары"
    РеализацияТоваровУслуг ||--|{ РеализацияТоваровУслуг_Товары : "содержит товары"
```
